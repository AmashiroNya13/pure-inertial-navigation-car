struct

typedef struct
{
    Ifx_P *port;     // 端口指针，例如 &MODULE_P33
    uint8  pinIndex; // 引脚索引
} IfxPort_Pin;

typedef struct
{
    Ifx_P            *port;      // 端口指针
    uint8             pinIndex;  // 引脚索引
    IfxPort_OutputIdx mode;      // 输出索引
    IfxPort_PadDriver padDriver; // pad driver 配置
} IfxPort_Pin_Config;

typedef struct
{
    IFX_CONST IfxPort_Pin *pin;      // 指向 pin map 中的引脚描述
    IfxPort_Mode           mode;     // 引脚模式
    IfxPort_PadDriver      padDriver; // pad driver
} IfxPort_Io_ConfigPin;

typedef struct
{
    uint32                size;     // pinTable 中元素个数
    IfxPort_Io_ConfigPin *pinTable; // 配置表首地址
} IfxPort_Io_Config;

// 与 TC38x 源文档差异：TC26B 当前 iLLD 没有 `IfxPort_LvdsConfig` 结构体。
// TC3xx 的统一 LVDS 配置结构在 TC26B 中被拆成了更具体的 LVDS High / Medium 函数参数。

--- GTM_TBU ---

TBU (Time Base Unit) 是 GTM 内部的全局时间基模块，提供 `TS0/TS1/TS2` 三路时间戳参考。它本身不是 PWM 或 ATOM/TOM Timer，不直接输出波形；更适合拿来做统一参考时间轴、模块间同步、硬件触发基准。

使用建议
1. 同一个 CMU 时钟只能保证多个 ATOM/TOM 通道“计数频率一致”，不能单独保证它们严格保持固定相位差。
2. 如果希望两个 PWM/Timer 在硬件上共享同一个时间参考，或希望比较/触发点绑定到同一参考时轴，应优先考虑 TBU/TS 机制。
3. 在底层 `IfxGtm_Atom_SomcControl_*` / `IfxGtm_Tom_*` 能看到 TS0/TS1/TS2 的比较模式；这类模式适合做“统一参考时刻 + 固定偏移”的硬件同步，而不是只靠软件顺序 start。
4. `IfxGtm_Atom_Agc_setTimeTrigger(...)`、`IfxGtm_Tom_Tgc_setTimeTrigger(...)` 用于把 AGC/TGC 的时间触发点绑定到某个 TBU 通道和值上；本工程当前高层 driver 没有继续封装这条能力，需要时通常要下沉到 iLLD 标准层直接配。

// ===== STM 关键注意事项 =====
// Timer API (IfxStm_Timer) 和 Compare API (IfxStm_initCompare) 是两套独立的 API，操作同一个硬件比较器，不可混用。
//
// Timer API 使用流程：
//   1. IfxStm_Timer_init  → 计算 comparatorShift/period，不配置 SRC（如果 isrPriority=0）
//   2. IfxStm_Timer_run    → 写入 CMP = STM + period，使能比较中断
//   3. ISR 中调用 IfxStm_Timer_acknowledgeTimerIrq → 清标志 + comparatorValue += period + 写 CMP
//   Timer API 内部维护 comparatorValue 和 period 作为软件状态，通过 acknowledge 递增。
//
// Compare API 使用流程：
//   1. IfxStm_initCompare  → 一步完成：配置窗口/偏移/SRC/中断使能，写入 CMP = current + ticks
//   2. ISR 中手动维护：
//        IfxStm_clearCompareFlag(stm, comparator);
//        IfxStm_updateCompare(stm, comparator, IfxStm_get(stm) + ticks);
//   Compare API 无内部软件状态，CMP 值完全由调用者控制。
//
// 混用后果：
//   - 调用 IfxStm_initCompare 设 CMP=current+1000 后，再调 IfxStm_Timer_run 会用 period(~10M)覆盖 CMP
//   - 在 Compare API 配置的 ISR 中调用 IfxStm_Timer_acknowledgeTimerIrq，每次加的却是 Timer 的 period，
//     与 Compare API 的 ticks 参数无关，导致中断间隔不等于预期的 ticks
//   - 总结：Timer_run 会覆盖 initCompare 写入的 CMP；acknowledgeTimerIrq 的递增量也来自 Timer 的 period，
//     不是 Compare API 的 ticks

    IfxSrc_Tos_dma  = 3   // DMA中断服务提供者
} IfxSrc_Tos;
